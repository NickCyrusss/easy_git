cmake_minimum_required(VERSION 3.22)
project(easy_git VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT UNIX OR APPLE)
  message(FATAL_ERROR "This first version supports Linux. The Git process runner uses POSIX APIs.")
endif()

option(EASY_GIT_BUILD_GUI "Build the desktop application" ON)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HTTP_JSON REQUIRED IMPORTED_TARGET libcurl json-c)
add_library(git_core src/git.cpp src/git_workflows.cpp src/app_services.cpp)
target_include_directories(git_core PUBLIC src)
target_link_libraries(git_core PUBLIC Threads::Threads PkgConfig::HTTP_JSON)
target_compile_options(git_core PRIVATE -Wall -Wextra -Wpedantic)

if(EASY_GIT_BUILD_GUI)
  include(FetchContent)
  # Use downloaded sources offline; otherwise FetchContent honors https_proxy.
  if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/glfw-3.4/CMakeLists.txt")
    set(FETCHCONTENT_SOURCE_DIR_GLFW "${CMAKE_SOURCE_DIR}/.deps/glfw-3.4")
  endif()
  if(EXISTS "${CMAKE_SOURCE_DIR}/.deps/imgui-1.92.5/imgui.h")
    set(FETCHCONTENT_SOURCE_DIR_IMGUI "${CMAKE_SOURCE_DIR}/.deps/imgui-1.92.5")
  endif()
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(glfw
    URL https://codeload.github.com/glfw/glfw/tar.gz/refs/tags/3.4
    URL_HASH SHA256=c038d34200234d071fae9345bc455e4a8f2f544ab60150765d7704e08f3dac01)
  FetchContent_Declare(imgui
    URL https://codeload.github.com/ocornut/imgui/tar.gz/refs/tags/v1.92.5
    URL_HASH SHA256=0eb50fe9aeba1a51f96b5843c7f630a32ed2e9362d693c61b87e4fa870cf826d)
  FetchContent_MakeAvailable(glfw imgui)
  find_package(OpenGL REQUIRED)
  add_library(imgui_backend
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(imgui_backend PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(imgui_backend PUBLIC glfw OpenGL::GL ${CMAKE_DL_LIBS})
  add_executable(easy_git src/main.cpp)
  # Embed RGBA pixels so the window icon also works when the binary is moved.
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS assets/easy_git.rgba)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/easy_git.rgba" icon_pixels HEX)
  string(REGEX REPLACE "(..)" "0x\\1," icon_pixels "${icon_pixels}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/app_icon.h"
    "static unsigned char app_icon_pixels[] = {${icon_pixels}};\nstatic_assert(sizeof(app_icon_pixels) == 64 * 64 * 4);\n")
  target_include_directories(easy_git PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(easy_git PRIVATE git_core imgui_backend)
  target_compile_options(easy_git PRIVATE -Wall -Wextra -Wpedantic)
  include(GNUInstallDirs)
  configure_file(assets/easy_git.desktop.in easy_git.desktop @ONLY)
  install(TARGETS easy_git RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/easy_git.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES assets/easy_git.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps)
  install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/easy-git RENAME copyright)
  set(CPACK_GENERATOR DEB)
  set(CPACK_PACKAGE_NAME easy-git)
  set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
  set(CPACK_PACKAGE_CONTACT "NickCyrusss <nickcyrus@163.com>")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Desktop Git client with commit graphs and optional AI commit messages")
  set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/NickCyrusss/easy_git")
  set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
  set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
  set(CPACK_DEBIAN_PACKAGE_SECTION devel)
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  # GLFW loads the X11/GL libraries dynamically, so shlibdeps cannot discover them.
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "git, libgl1, libx11-6, libxrandr2, libxinerama1, libxcursor1, libxi6, libxxf86vm1")
  set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "zenity, fonts-dejavu-core, fonts-noto-cjk")
  set(CPACK_STRIP_FILES ON)
  include(CPack)
endif()

  add_executable(git_workflows_test tests/git_workflows_test.cpp)
  target_link_libraries(git_workflows_test PRIVATE git_core)
  add_test(NAME git_workflows COMMAND git_workflows_test)
  add_executable(app_services_test tests/app_services_test.cpp)
  target_link_libraries(app_services_test PRIVATE git_core)
  add_test(NAME settings_and_ai COMMAND app_services_test)
  if(EASY_GIT_BUILD_GUI)
    add_executable(ui_state_test tests/ui_state_test.cpp)
    target_link_libraries(ui_state_test PRIVATE git_core imgui_backend)
    add_test(NAME repository_tabs COMMAND ui_state_test)
  endif()
endif()
